const rangeStates = new Map();
function sidewaysBox(m) {
  const id = JSON.stringify([m.server, m.login, m.symbol]);
  const rows = m.candles.slice(-9);
  if (m.count !== 0 || rows.length < 9 || !(m.atr > 0) ||
      rows.some((r,i) => !Array.isArray(r) || !r.slice(0,5).every(Number.isFinite) ||
        r.length < 5 || (i && r[0] - rows[i-1][0] !== 300))) {
    rangeStates.delete(id);
    return null;
  }
  const [time,,,,close] = rows.at(-1);
  let state = rangeStates.get(id);
  if (state && time === state.time) return state.box;
  if (!state || time !== state.time + 300) state = {box:null, since:0};
  let box = state.box;
  const prior = rows.slice(0,-1);
  if (!box && prior[0][0] >= state.since) {
    const high = Math.max(...prior.map(r => r[2]));
    const low = Math.min(...prior.map(r => r[3]));
    const travel = prior.slice(1).reduce((sum,r,i) => sum + Math.abs(r[4]-prior[i][4]), 0);
    const drift = Math.abs(prior.at(-1)[4]-prior[0][4]);
    if (high > low && high-low <= 3*m.atr && drift <= 0.35*travel) box = {high,low};
  }
  if (box && (close > box.high || close < box.low)) {
    box = null;
    state.since = time;
  }
  rangeStates.set(id, {...state, box, time});
  return box;
}
